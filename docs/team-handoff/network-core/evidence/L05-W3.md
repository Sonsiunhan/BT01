## Chi tiết Triển khai Kỹ thuật L05-W3

### 1. Unified Independent Deadline Scheduler (`deadline-scheduler.ts`)
```typescript
export class DeadlineScheduler {
  private readonly registeredRooms = new Set<string>();
  private readonly deadlines = new Map<string, Map<string, { deadline: ScheduledDeadline; timer?: NodeJS.Timeout }>>();
  private readonly roomTimers = new Map<string, NodeJS.Timeout>();
  private readonly now: () => number;

  startRoom(roomId: string, onTick: () => void, intervalMs = 1_000): void {
    this.registerRoom(roomId);
    if (this.roomTimers.has(roomId)) return;
    const timer = setInterval(() => {
      try { onTick(); } catch {}
    }, intervalMs);
    timer.unref?.();
    this.roomTimers.set(roomId, timer);
  }

  schedule(deadline: ScheduledDeadline): void {
    this.registerRoom(deadline.roomId);
    // Bounded deadline queue with unref'd timer and immediate onDue dispatch
    const delay = Math.max(0, deadline.dueAt - this.now());
    const timer = setTimeout(() => {
      this.cancel(deadline.roomId, deadline.key);
      try { void deadline.onDue(); } catch {}
    }, delay);
    timer.unref?.();
    roomMap.set(deadline.key, { deadline, timer });
  }
}
```

### 2. Terminal Precedence trong Surrender (`match.manager.ts`)
```typescript
  surrender(room: RoomDetail, actor: MatchActor, stateVersion: number): MatchSnapshot {
    const match = this.getMatch(room);
    this.advance(match, true);
    // Strict terminal timeout precedence: timeout expires before command can commit
    if (match.status === "FINISHED" || match.status === "ABORTED") {
      return this.snapshot(match);
    }
    const player = this.player(match, actor.userId);
    this.assertVersion(match, stateVersion);
    if (match.status !== "PLAYING") throw this.invalidState("surrender", match.status);
    this.finish(match, player.side === "BLUE" ? "RED" : "BLUE", "SURRENDER");
    this.emit(match, "PLAYER_SURRENDERED");
    this.emit(match, "MATCH_FINISHED");
    return this.snapshot(match);
  }
```

### 3. Fencing cho Bot Adjudication (`match.manager.ts`)
```typescript
  finishBotLimit(
    room: RoomDetail,
    winner: Side,
    adjudication: MatchBotAdjudication,
    expectedStateVersion?: number,
    expectedMatchId?: string,
    epoch?: number
  ): MatchSnapshot {
    const match = this.getMatch(room);
    if (match.playMode !== "BOT") throw new AppError("CONFLICT", "Chỉ trận Đấu chương trình mới có thể phân định giới hạn.", 409, false, "INVALID");
    if (expectedMatchId !== undefined && expectedMatchId !== match.matchId) {
      throw new AppError("CONFLICT", "Lệnh Bot thuộc ván đấu cũ.", 409, true, "RECOVERABLE", { reason: "STALE_MATCH_ID" });
    }
    if (expectedStateVersion !== undefined && expectedStateVersion !== match.stateVersion) {
      throw new AppError("CONFLICT", "Trạng thái bàn cờ đã thay đổi.", 409, true, "RECOVERABLE", { reason: "STALE_VERSION" });
    }
    if (epoch !== undefined && epoch !== match.stateVersion) {
      throw new AppError("CONFLICT", "Epoch của Bot không khớp trạng thái hiện tại.", 409, true, "RECOVERABLE", { reason: "STALE_EPOCH" });
    }
    if (match.status !== "PLAYING") return this.snapshot(match);
    this.finish(match, winner, adjudication.reason === "LIMIT_EXACT_TIE" ? "BOT_LIMIT_EXACT_TIE" : "BOT_LIMIT_CRITERIA");
    match.botAdjudication = adjudication;
    this.emit(match, "MATCH_FINISHED");
    return this.snapshot(match);
  }
```
