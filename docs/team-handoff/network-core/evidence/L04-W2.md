### 2. Monotonic Reconciler với Gap Detection và Stale ACK Guard (`reconciler.ts`)
```typescript
export function reconcileDomainEvent(
  reconciler: ReconcilerState,
  event: MatchDomainEventEnvelope,
  applyEventToState: (current: CanonicalMatchSnapshot, event: MatchDomainEventEnvelope) => CanonicalMatchSnapshot
): ReconcileResult {
  if (event.scope !== reconciler.scope || event.targetId !== reconciler.targetId) {
    return { applied: false, gapDetected: false, reason: "SCOPE_MISMATCH" };
  }
  // Stale or duplicate event
  if (event.sequence <= reconciler.watermarkSequence) {
    return { applied: false, gapDetected: false, reason: "DUPLICATE_EVENT" };
  }
  // Gap detected
  if (event.sequence > reconciler.watermarkSequence + 1) {
    return { applied: false, gapDetected: true, reason: "GAP_DETECTED" };
  }
  ...
}
```
