### 1. Per-Match Serialized Bounded Lane (`match-lane.ts`)
```typescript
export class MatchLane {
  readonly roomId: string;
  private readonly maxQueueDepth: number;
  private readonly queue: Array<{ task: LaneTask; resolve: (v: any) => void; reject: (e: any) => void; enqueuedAt: number }>;
  private isProcessing = false;
  private isUnknownLocked = false;

  async enqueue<T>(task: LaneTask<T>): Promise<T> {
    if (this.isUnknownLocked) {
      throw new AppError("CONFLICT", "Trận đấu bị khóa tạm thời do giao dịch chưa xác định", 409, true, "RECOVERABLE", { reason: "LANE_UNKNOWN_LOCKED" });
    }
    if (this.queue.length >= this.maxQueueDepth) {
      throw new AppError("RATE_LIMITED", "Hàng đợi lệnh của trận đấu đã đầy. Vui lòng thử lại sau.", 429, true, "RECOVERABLE", { reason: "LANE_QUEUE_FULL" });
    }
    return new Promise<T>((resolve, reject) => {
      this.queue.push({ task, resolve, reject, enqueuedAt: Date.now() });
      void this.processQueue();
    });
  }
}
```
